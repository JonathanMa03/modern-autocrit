\begin{frame}{Why this tool is needed}
Eligibility criteria determine who can enter a trial, yet important differences are buried in long protocols and inconsistent prose.
\begin{itemize}
\item Study teams compare criteria for protocol design, feasibility, competitive intelligence, evidence synthesis, and amendment review.
\item Keyword search finds shared words; ordinary embeddings often find shared topics. Neither guarantees shared eligibility meaning.
\item Users need to know whether criteria are equivalent, narrower, overlapping, contradictory, related, or incomparable---with evidence and uncertainty.
\end{itemize}
\begin{block}{Practical goal}Select a criterion, retrieve comparable criteria across protocols, inspect the differences, and trace the result back to its source.\end{block}
\end{frame}

\begin{frame}{From AutoCriteria to Modern AutoCrit}
\begin{columns}[T,onlytextwidth]\column{.46\textwidth}
\textbf{AutoCriteria established}
\begin{itemize}\item LLM-assisted conversion of eligibility prose into structured criteria\item a scalable starting point for protocol preprocessing\end{itemize}
\column{.51\textwidth}\textbf{Modern AutoCrit adds}
\begin{itemize}\item NCT, pasted-text, and full-protocol PDF ingestion\item atomic thresholds, negation, timing, repetition, exceptions, and AND/OR scope\item provenance, stable IDs, validation, human correction, and resumable jobs\item versioned Excel and validated JSON exports\end{itemize}
\end{columns}
\begin{block}{Engineering change}The OpenAI SDK and small testable modules replaced LangChain/LangGraph and the original long script, making state, retries, validation, and provenance explicit.\end{block}
\source{Datta et al., AutoCriteria (2023); github.com/JonathanMa03/modern-autocrit}
\end{frame}

\begin{frame}{The gap MetricSpace addresses}
\begin{tabularx}{\textwidth}{@{}X X X@{}}\toprule Pair & Embedding problem & Clinical relation \\\midrule
Age $\geq18$ vs. age $<18$ & nearly identical wording & disjoint populations \\
No cardiovascular disease vs. controlled hypertension permitted & same clinical topic & exception changes scope \\
No prior chemotherapy vs. none within 6 months & high lexical overlap & temporal rule differs \\

\begin{block}{Three questions}Where do conventional embeddings fail? Can reviewed hard cases improve a dense encoder? Can an agent choose the right tools, quantify uncertainty, and escalate safely?\end{block}
\end{frame}

\begin{frame}{Protocol corpus: two complementary areas}
\begin{columns}[T,onlytextwidth]\column{.46\textwidth}
\begin{block}{50 oncology trials}Biomarkers, stage, histology, prior therapy, washout periods, performance status, organ function, and nested exceptions.\end{block}
\column{.46\textwidth}
\begin{block}{50 cardiovascular trials}Heart failure, atrial fibrillation, coronary/valvular disease, hypertension, stroke, thrombosis, PAD, and aneurysm.\end{block}\end{columns}
\begin{itemize}\item Target: approximately 2,000 human-reviewed atomic criteria.\item Chia may supplement terminology and examples, but does not supply gold pairwise labels.\item Every atom retains NCT ID, source version, parent statement, context, review, and validation.\end{itemize}
\source{Kury et al., Chia, Scientific Data 7, 281 (2020).}
\end{frame}

\begin{frame}{Interpretable relationships, not an opaque score}
\begin{columns}[T,onlytextwidth]\column{.56\textwidth}
\begin{itemize}\item \pill{Equivalent} same operational rule\item \pill{More restrictive} one permits a subset\item \pill{Partially overlapping} shared but non-nested\item \pill{Contradictory / disjoint} cannot both hold\item \pill{Related, not comparable} shared topic, different operation\item \pill{Unrelated} no meaningful relationship\item \pill{Ambiguous} insufficient context\end{itemize}
\column{.40\textwidth}\begin{block}{Why this matters}Similarity can rank candidates, but users see a clinical relationship, evidence, and calibrated uncertainty rather than a weighted mixture with unclear meaning.\end{block}\end{columns}
\end{frame}

The agent selects tools, reconciles disagreements, records its trace, and abstains when evidence is insufficient. Exact cases are not forced through an LLM; ambiguous cases are not hidden behind cosine similarity.
\end{frame}

\begin{frame}{Human steering becomes representation learning}
\begin{enumerate}\item The agent retrieves pairs and assembles structural and source evidence.\item An LLM proposes a relationship; exact checks or a critic challenge conflicts.\item A human accepts, corrects, or marks ambiguity. The human decision is authoritative.\item Reviewed pairs form positives, hard negatives, and ordered anchors.\item One open dense encoder is fine-tuned offline and calibrated on held-out data.\item Later review prioritizes uncertainty and disagreement, compared with random sampling.\end{enumerate}
\begin{block}{Contribution}A reviewed protocol dataset, clinically difficult negatives, agentic labeling workflow, and controlled evaluation---not merely checkpoint fine-tuning.\end{block}
\end{frame}

\begin{frame}{Evaluation plan}
\begin{tabularx}{\textwidth}{@{}p{.27\textwidth}X@{}}\toprule Component & Measures \\\midrule
Retrieval & precision/recall@$k$, mean reciprocal rank, anchor-ordering accuracy \\
Relationship prediction & macro-F1, class errors, confusion matrix, order consistency \\
Uncertainty & calibration, risk--coverage, abstention and escalation errors \\
Agent usefulness & correction rate, conflict detection, reviewer time, latency, and cost \\
Statistics & trial-family bootstrap intervals, paired comparisons, and ablations \\\bottomrule\end{tabularx}
\vspace{.5em}Trial families are split before pair construction; amendments and shared templates stay together. Held-out labels remain unavailable to agent retrieval.
\end{frame}

\begin{frame}{What we will build and how}
\begin{columns}[T,onlytextwidth]\column{.48\textwidth}\textbf{Working tool}
\begin{itemize}\item cross-protocol criterion retrieval\item field-level and contextual differences\item relationship and uncertainty\item traceable source evidence\item correction and escalation\end{itemize}
\column{.48\textwidth}\textbf{Execution}
\begin{enumerate}\item calibrate on 10 trials\item extract/review the 100-trial cohort\item measure embedding failures and label pairs\item fine-tune and calibrate one encoder\item integrate agent, critic, audit, and abstention\end{enumerate}\end{columns}
\begin{block}{Safeguards}Human-reviewed inputs, blinded test labels, trial-family splits, and no automated patient screening, causal claims, or opaque hybrid score.\end{block}
\end{frame}

\begin{frame}{Take-home proposal}
\begin{block}{MetricSpace}An agentic clinical data-science tool combining reviewed protocol structure, exact checks, retrieval, embeddings, contextual LLM reasoning, and human steering to compare eligibility criteria with interpretable relationships and calibrated uncertainty.\end{block}
\vspace{1em}\begin{center}\large Modern AutoCrit supplies traceable atoms.\\[.45em]Human review supplies trustworthy relationships.\\[.45em]MetricSpace learns and operationalizes the comparison.\end{center}

\end{frame}
